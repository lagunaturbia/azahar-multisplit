cmake_minimum_required(VERSION 3.18)
project(AzaharMultisplit LANGUAGES CXX)

add_library(azahar_multisplit_libretro SHARED multisplit.cpp gpu_compositor.cpp vk_compositor.cpp)
set_target_properties(azahar_multisplit_libretro PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
target_compile_features(azahar_multisplit_libretro PRIVATE cxx_std_17)
find_path(AZAHAR_VULKAN_HEADERS NAMES vulkan/vulkan.h REQUIRED)
target_include_directories(azahar_multisplit_libretro PRIVATE "${AZAHAR_VULKAN_HEADERS}" "${CMAKE_CURRENT_SOURCE_DIR}")

if(WIN32)
    enable_language(RC)
    set(AZAHAR_ENGINE_DLL "" CACHE FILEPATH "Patched Azahar Libretro 2126.2 Windows x64 DLL")
    if(NOT EXISTS "${AZAHAR_ENGINE_DLL}")
        message(FATAL_ERROR "Set AZAHAR_ENGINE_DLL to the patched azahar_libretro.dll")
    endif()
    file(TO_CMAKE_PATH "${AZAHAR_ENGINE_DLL}" AZAHAR_ENGINE_RESOURCE_PATH)
    set(resource "${CMAKE_CURRENT_BINARY_DIR}/azahar_engine.rc")
    file(WRITE "${resource}" "101 RCDATA \"${AZAHAR_ENGINE_RESOURCE_PATH}\"\n")
    target_sources(azahar_multisplit_libretro PRIVATE "${resource}")
    set_source_files_properties("${resource}" PROPERTIES OBJECT_DEPENDS "${AZAHAR_ENGINE_DLL}")
    target_link_libraries(azahar_multisplit_libretro PRIVATE opengl32 gdi32 user32)
    if(MINGW)
        target_link_options(azahar_multisplit_libretro PRIVATE -static)
    endif()
else()
    find_library(EGL_LIBRARY NAMES EGL libEGL.so.1 REQUIRED)
    target_link_libraries(azahar_multisplit_libretro PRIVATE ${CMAKE_DL_LIBS} "${EGL_LIBRARY}")
    option(MULTISPLIT_BUILD_SMOKE "Build the Libretro/EGL smoke test" OFF)
    if(MULTISPLIT_BUILD_SMOKE)
        add_executable(multisplit_smoke smoke.cpp)
        target_compile_features(multisplit_smoke PRIVATE cxx_std_17)
        target_link_libraries(multisplit_smoke PRIVATE ${CMAKE_DL_LIBS} "${EGL_LIBRARY}")
    endif()
endif()
